(define (problem hanoi1)
    (:domain hanoi)
    (:objects k1 k2 k3 k4 k5 x y z)
    (:init
        (palik x)
        (palik y)
        (palik z)
        (na_paliku x k1)
        (na_paliku x k2)
        (na_paliku x k3)
        (na_paliku x k4)
        (na_paliku x k5)
        (smaller k1 k2)
        (smaller k1 k3)
        (smaller k1 k4)
        (smaller k1 k5)
        (smaller k2 k3)
        (smaller k2 k4)
        (smaller k2 k5)
        (smaller k3 k4)
        (smaller k3 k5)
        (smaller k4 k5)
    )
    (:goal 
        (and
            (na_paliku z k1)
            (na_paliku z k2)
            (na_paliku z k3)
            (na_paliku z k4)
            (na_paliku z k5)
    ))
)
